/************************************************* * CONFIG *************************************************/ const DIMENSIONS = { selfAwareness: { label: "خودشناسی", importance: 1.4 }, habits: { label: "عادت‌ها", importance: 1.2 }, mindset: { label: "ذهنیت", importance: 1.3 }, goals: { label: "هدف‌گذاری", importance: 1.1 } }; const QUESTIONS = [ { text: "احساساتم را به‌خوبی می‌شناسم", dimension: "selfAwareness" }, { text: "الگوهای رفتاری خودم را می‌فهمم", dimension: "selfAwareness" }, { text: "عادت‌های روزانه‌ی من پایدار هستند", dimension: "habits" }, { text: "می‌توانم عادت بد را کنترل کنم", dimension: "habits" }, { text: "در شرایط سخت ذهنم آرام می‌ماند", dimension: "mindset" }, { text: "نگرش مثبتی به آینده دارم", dimension: "mindset" }, { text: "اهدافم واضح و مشخص هستند", dimension: "goals" }, { text: "برای اهدافم برنامه دارم", dimension: "goals" } ]; /************************************************* * STATE *************************************************/ let state = { index: 0, answers: [] }; /************************************************* * INIT *************************************************/ renderQuestion(); /************************************************* * RENDER QUESTION *************************************************/ function renderQuestion() { const q = QUESTIONS[state.index]; const cardArea = document.getElementById("cardArea"); cardArea.innerHTML = `

${q.text}

${[1,2,3,4,5].map(v => `` ).join("")}
`; updateProgress(); } /************************************************* * ANSWER *************************************************/ function answer(value) { state.answers.push({ ...QUESTIONS[state.index], value }); state.index++; if (state.index < QUESTIONS.length) { renderQuestion(); } else { analyzeAssessment(); } } /************************************************* * PROGRESS *************************************************/ function updateProgress() { const percent = ((state.index) / QUESTIONS.length) * 100; document.getElementById("progressFill").style.width = percent + "%"; } /************************************************* * ANALYSIS *************************************************/ function analyzeAssessment() { const data = {}; Object.keys(DIMENSIONS).forEach(k => { data[k] = { total: 0, count: 0 }; }); state.answers.forEach(a => { data[a.dimension].total += a.value; data[a.dimension].count++; }); Object.keys(DIMENSIONS).forEach(k => { const avg = data[k].total / data[k].count; const score = Math.round((avg / 5) * 100); data[k].score = score; data[k].priority = DIMENSIONS[k].importance * (100 - score); const interp = interpretDimension(score); data[k].state = interp.state; data[k].stateLabel = interp.label; data[k].coachHint = interp.coachMessage; }); showResult(data); } /************************************************* * COACH INTERPRETATION (FINAL) *************************************************/ function interpretDimension(score) { if (score < 40) { return { state: "STABILIZE", label: "نیاز به تثبیت", coachMessage: "اول آرام‌سازی و بازسازی پایه‌ها" }; } if (score < 70) { return { state: "GROW", label: "بهترین اهرم رشد ✅", coachMessage: "اینجا بیشترین بازده رشد را داری" }; } return { state: "MAINTAIN", label: "نقطه قوت", coachMessage: "حفظ و بهینه‌سازی" }; } /************************************************* * PICK COACH FOCUS *************************************************/ function pickCoachFocus(data) { const grow = Object.keys(DIMENSIONS) .filter(k => data[k].state === "GROW") .sort((a,b)=>data[b].priority - data[a].priority); if (grow.length) return grow[0]; const stabilize = Object.keys(DIMENSIONS) .filter(k => data[k].state === "STABILIZE") .sort((a,b)=>data[b].priority - data[a].priority); return stabilize[0]; } /************************************************* * SHOW RESULT *************************************************/ function showResult(data) { const cardArea = document.getElementById("cardArea"); const focusKey = pickCoachFocus(data); let html = `

📊 نتیجه ارزیابی

`; Object.keys(DIMENSIONS).forEach(k => { html += `
${DIMENSIONS[k].label}: ${data[k].score}٪
${data[k].stateLabel} – ${data[k].coachHint}
`; }); html += `
🧠 تمرکز مربی:
${DIMENSIONS[focusKey].label}
${data[focusKey].coachHint}
`; cardArea.innerHTML = html; document .getElementById("enterProgramBtn") .addEventListener("click", () => { startCoachProgram(focusKey, data); }); } /************************************************* * NEXT STEP HOOK *************************************************/ function startCoachProgram(focusKey, data) { console.log("Coach program started:", focusKey, data); // مرحله بعد: // ✅ گفتگوی پویا با Coach // ✅ برنامه ۷ روزه // ✅ تمرین‌های روزانه }
